// 前端音频采集 (Vue3)
audioContext = new AudioContext({ sampleRate: 16000 });
mediaStream = await navigator.mediaDevices.getUserMedia({
    audio: {
        sampleRate: 16000,     // 16kHz 采样率
        channelCount: 1,       // 单声道
        echoCancellation: true // 回声消除
    }
});

scriptProcessor = audioContext.createScriptProcessor(4096, 1, 1);
scriptProcessor.onaudioprocess = (event) => {
    const input = event.inputBuffer.getChannelData(0); // Float32Array
    
    // 转换为 16位 PCM
    const pcmData = new Int16Array(input.length);
    for (let i = 0; i < input.length; i++) {
        let sample = Math.max(-1, Math.min(1, input[i]));
        pcmData[i] = sample < 0 ? sample * 0x8000 : sample * 0x7FFF;
    }
    
    // Base64 编码并发送
    const base64Audio = btoa(String.fromCharCode.apply(null, pcmData));
    websocket.send(JSON.stringify({
        type: 'chunk',
        audio: base64Audio,
        format: 'pcm',
        sample_rate: 16000
    }));
};

3. 后端处理流程

# Python 后端处理 (rest.py)
@sock.route('/ws/recognize')
def ws_recognize(ws):
    session = RealtimeSpeechSession()
    
    while True:
        message = ws.receive()
        data = json.loads(message)
        
        if data['type'] == 'chunk':
            # 1. 解码音频数据
            audio_data = base64.b64decode(data['audio'])
            audio_array = np.frombuffer(audio_data, dtype=np.int16)
            audio_array = audio_array.astype(np.float32) / 32768.0
            
            # 2. 添加到缓冲区
            session.add_audio_chunk(audio_array, 'pcm', 16000)
            
            # 3. 实时识别
            partial_result = session.get_partial_result()
            if partial_result:
                ws.send(json.dumps({
                    "type": "partial",
                    "text": partial_result["text"],
                    "confidence": partial_result["confidence"]
                }))
            
            # 4. 句子完成检测
            if session.check_sentence_complete():
                final_text = session.finalize_current_sentence()
                ws.send(json.dumps({
                    "type": "final", 
                    "text": final_text
                }))

句子完成检测算法

def check_sentence_complete(self):
    if not self.current_sentence:
        return False
    
    silence_duration = time.time() - self.last_activity_time
    
    # 短句需要更长确认时间
    if len(self.current_sentence) < 2:
        return silence_duration > 2.25  # 1.5 * 1.5
    
    # 有标点符号的句子快速确认
    if any(end in self.current_sentence for end in ['。', '！', '？', '.', '!', '?']):
        return silence_duration > 0.8
    
    # 长句子较快确认
    if len(self.current_sentence) > 15:
        return silence_duration > 1.0
    
    # 标准确认时间
    return silence_duration > 1.5

异常处理机制

// 前端连接异常处理
websocket.onclose = (event) => {
    wsConnected.value = false;
    stopHeartbeat();
    
    // 如果正在录音，自动停止
    if (isRealtimeRecording.value) stopRealtime();
    
    // 启动重连逻辑
    scheduleReconnect();
};

function scheduleReconnect() {
    if (reconnectAttempts >= 5) return; // 最多重连5次
    
    const delay = Math.min(15000, 1000 * Math.pow(2, reconnectAttempts));
    reconnectAttempts++;
    
    setTimeout(() => {
        if (!wsConnected.value && !wsConnecting.value) {
            connectWS();
        }
    }, delay);
}

优雅断开流程

⚠️ 断开连接最佳实践：
  1. 发送end消息：通知后端处理最终结果
  2. 等待final响应：确保最后一句话被处理
  3. 清理音频资源：停止录音，关闭AudioContext
  4. 关闭WebSocket：使用正确的关闭码
// 优雅断开实现
function disconnectWS() {
    if (websocket) {
        // 1. 如果正在录音，先发送结束信号
        if (isRealtimeRecording.value && websocket.readyState === WebSocket.OPEN) {
            websocket.send(JSON.stringify({ type: 'end' }));
        }
        
        // 2. 正常关闭连接
        websocket.close(1000, '用户主动断开');
        websocket = null;
    }
    
    // 3. 清理所有资源
    cleanupAudio();
    stopHeartbeat();
}

🔧 问题排查指南

常见问题诊断

🚨 问题1：识别结果为空（如用户遇到的问题）

症状：音频块接收正常，但实时和最终结果都是0

诊断步骤：

  1. 查看 logs/audio_processing.log 音频质量统计
  2. 检查 logs/recognition_results.log 识别过程
  3. 确认ModelScope模型调用是否返回结果

可能原因：

  • 音频RMS值过低（< 0.005）被判断为静音
  • 音频长度不足300ms被跳过
  • 识别间隔限制（< 500ms）
  • ModelScope返回空结果或解析失败
💡 解决方案：
  1. 调整音频阈值：将有效音频阈值从0.005降至0.001
  2. 缩短最小长度：最小音频长度从300ms降至100ms
  3. 增加调试日志：使用增强版的识别函数查看详细过程
  4. 测试环境：在安静环境下，靠近麦克风，提高音量测试
