```python
def diff_detailed(self, new_text: str) -> List[Dict[str, Any]]:
    """
    ✅ P1-1 新增：使用编辑距离算法精确计算差异。

    返回详细的编辑操作列表：
    [
        {'type': 'delete', 'text': '好', 'position': 5, 'length': 1},
        {'type': 'insert', 'text': '不错', 'position': 5, 'length': 2},
        {'type': 'replace', 'old_text': 'A', 'new_text': 'B', 'position': 3, ...}
    ]

    示例：
        原文: "今天天气真好"
        新文: "今天的天气真不错"
        结果: [
            {'type': 'insert', 'text': '的', 'position': 2, 'length': 1},
            {'type': 'replace', 'old_text': '好', 'new_text': '不错', 'position': 6, ...}
        ]
    """
    if not self.confirmed_text and not new_text:
        return []

    matcher = SequenceMatcher(None, self.confirmed_text, new_text)
    changes = []

    for tag, i1, i2, j1, j2 in matcher.get_opcodes():
        if tag == 'delete':
            changes.append({
                'type': 'delete',
                'text': self.confirmed_text[i1:i2],
                'position': i1,
                'length': i2 - i1,
            })
        elif tag == 'insert':
            changes.append({
                'type': 'insert',
                'text': new_text[j1:j2],
                'position': i1,
                'length': j2 - j1,
            })
        elif tag == 'replace':
            changes.append({
                'type': 'replace',
                'old_text': self.confirmed_text[i1:i2],
                'new_text': new_text[j1:j2],
                'position': i1,
                'old_length': i2 - i1,
                'new_length': j2 - j1,
            })

    return changes
```

**位置 3**: 增强 `apply_correction()` 方法 (line 102-129)

**变更内容**:
```python
def apply_correction(self, text: str) -> Dict[str, str | int | List]:
    """
    应用纠错并返回详细差异。

    ✅ P1-1 增强：返回 snapshot 中包含 'changes' 字段，提供详细的编辑操作列表。
    """
    # 使用新的详细差异算法
    changes = self.diff_detailed(text)

    if self.history is None:
        self.history = []

    # 记录历史（向后兼容旧格式）
    removed, appended = self.diff(text)
    if removed:
        self.history.append({"removed": removed})
    if appended:
        self.history.append({"appended": appended})

    # 更新状态
    self.confirmed_text = text
    self.candidate_text = ""
    self.revision_id += 1

    # 返回包含详细差异的 snapshot
    snapshot = self.snapshot()
    snapshot['changes'] = changes  # ✅ 新增详细差异信息
    return snapshot
```

**影响**:
- 纠正提示更精确，能识别中间插入、删除、替换操作
- 前端可通过 `data.text_state.changes` 获取详细的编辑列表
- 向后兼容旧的 `diff()` 方法

**示例对比**:

**变更内容**:
```javascript
function processAudioFrame(event) {
    if (!isStreaming || !websocket || websocket.readyState !== WebSocket.OPEN) {
        return;
    }

    const input = event.inputBuffer.getChannelData(0);

    // ✅ P1-2 修复：累积音频到缓冲区，达到 CHUNK_STRIDE 时再发送
    // 原方案：每 4096 samples (256ms) 发送一次
    // 新方案：累积到 9600 samples (600ms) 再发送，对齐 FunASR 官方推荐
    audioBuffer.push(...input);

    // 计算 RMS（使用当前输入的音频）
    const rms = Math.sqrt(input.reduce((sum, sample) => sum + sample * sample, 0) / input.length);
    document.getElementById('rmsDisplay').textContent = `RMS: ${rms.toFixed(4)} (缓冲: ${audioBuffer.length}/${CHUNK_STRIDE})`;

    // 当累积到 CHUNK_STRIDE 时发送
    while (audioBuffer.length >= CHUNK_STRIDE) {
        // 取出 CHUNK_STRIDE 长度的音频
        const chunk = audioBuffer.slice(0, CHUNK_STRIDE);
        audioBuffer = audioBuffer.slice(CHUNK_STRIDE);

        // 转换为 Int16 PCM
        const pcm = new Int16Array(chunk.length);
        for (let i = 0; i < chunk.length; i++) {
            const s = Math.max(-1, Math.min(1, chunk[i]));
            pcm[i] = s < 0 ? s * 0x8000 : s * 0x7fff;
        }

        // 打包为 ArrayBuffer
        const buffer = new ArrayBuffer(pcm.length * 2);
        const view = new DataView(buffer);
        for (let i = 0; i < pcm.length; i++) {
            view.setInt16(i * 2, pcm[i], true);
        }

        // Base64 编码
        const bytes = new Uint8Array(buffer);
        const base64 = btoa(String.fromCharCode.apply(null, bytes));

        const message = {
            type: 'chunk',
            format: 'pcm',
            sample_rate: SAMPLE_RATE,
            audio: base64,
            rms: rms.toFixed(4),
            ts: Date.now(),
            chunk_size: chunk.length  // ✅ 新增：标识分块大小
        };

        try {
            websocket.send(JSON.stringify(message));
        } catch (error) {
            appendLog('error', `发送音频失败: ${error.message}`);
            break;  // 发送失败时停止处理
        }
    }
}
```
